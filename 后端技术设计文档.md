### 5.1 刷题 Agent（PracticeAgent）

```python
# agents/practice_agent.py
from typing import TypedDict, List, Optional, Annotated
from langgraph.graph import StateGraph, END
from langgraph.graph.message import add_messages

class DifficultyLevel:
    BEGINNER = 1
    ELEMENTARY = 2
    INTERMEDIATE = 3
    ADVANCED = 4
    EXPERT = 5

class PracticeState(TypedDict):
    # 会话信息
    session_id: int
    user_id: int
    topic: str

    # 当前题目
    current_question: Optional[str]
    question_type: Optional[str]           # short_answer / choice
    reference_answer: Optional[str]
    knowledge_points: Optional[List[str]]

    # 用户交互
    user_answer: Optional[str]

    # 评估结果
    score: Optional[float]
    is_correct: Optional[bool]
    feedback: Optional[str]                 # 完整解析（Markdown）

    # 难度管理
    difficulty_level: int                   # 当前档位 1-5
    consecutive_correct: int
    consecutive_wrong: int
    question_index: int

    # 流式输出控制
    stream_buffer: Optional[str]            # 当前流式产出缓存

    # 流程控制
    next_action: str                        # generate_question | wait_answer |
                                            # evaluate | feedback | end
    error: Optional[str]
```

**关键节点实现**：

```python
class PracticeAgent:
    def __init__(self, llm_client: UnifiedLLMClient, practice_repo: PracticeRepo):
        self.llm = llm_client
        self.repo = practice_repo
        self.graph = self._build_graph()

    def _build_graph(self) -> StateGraph:
        workflow = StateGraph(PracticeState)

        workflow.add_node("validate_topic", self.validate_topic)
        workflow.add_node("generate_question", self.generate_question)
        workflow.add_node("wait_answer", self.wait_answer)
        workflow.add_node("evaluate", self.evaluate)
        workflow.add_node("feedback", self.feedback)
        workflow.add_node("adjust_difficulty", self.adjust_difficulty)
        workflow.add_node("wait_next", self.wait_next)

        workflow.set_entry_point("validate_topic")

        workflow.add_conditional_edges("validate_topic", self._after_validate, {
            "generate_question": "generate_question",
            "error": END,
        })
        workflow.add_edge("generate_question", "wait_answer")
        workflow.add_edge("wait_answer", "evaluate")
        workflow.add_edge("evaluate", "feedback")
        workflow.add_edge("feedback", "adjust_difficulty")
        workflow.add_edge("adjust_difficulty", "wait_next")
        workflow.add_edge("wait_next", "generate_question")

        return workflow.compile(checkpointer=self._checkpointer)

    async def validate_topic(self, state: PracticeState) -> PracticeState:
        """校验主题是否属于技术面试领域"""
        prompt = VALIDATE_TOPIC_PROMPT.format(topic=state["topic"])
        result = await self.llm.chat(prompt, response_format={"type": "json_object"})
        data = json.loads(result)

        if data.get("is_valid", False):
            state["next_action"] = "generate_question"
        else:
            state["error"] = "请输入技术面试相关主题"
            state["next_action"] = "error"
        return state

    async def generate_question(self, state: PracticeState) -> PracticeState:
        """生成题目（流式）"""
        session = await self.repo.get_session(state["session_id"])
        history_summary = await self._build_history_summary(session)

        prompt = GENERATE_QUESTION_PROMPT.format(
            topic=state["topic"],
            difficulty=DIFFICULTY_LABELS[state["difficulty_level"]],
            question_index=state["question_index"] + 1,
            history_summary=history_summary,
        )
        # 流式产出存入 stream_buffer，由 SSE emitter 消费
        state["stream_buffer"] = ""
        state["next_action"] = "wait_answer"
        state["question_index"] += 1
        return state

    async def evaluate(self, state: PracticeState) -> PracticeState:
        """评分 + 生成解析"""
        prompt = EVALUATE_PROMPT.format(
            topic=state["topic"],
            question=state["current_question"],
            reference_answer=state["reference_answer"],
            user_answer=state["user_answer"],
        )
        result = await self.llm.chat(prompt, response_format={"type": "json_object"})
        eval_data = json.loads(result)

        state["score"] = eval_data["score"]
        state["is_correct"] = eval_data["score"] >= 60
        state["feedback"] = eval_data["feedback"]
        state["next_action"] = "feedback"
        return state

    async def adjust_difficulty(self, state: PracticeState) -> PracticeState:
        """难度自适应：连续答对 3 题升一档，连续答错 2 题降一档"""
        if state["is_correct"]:
            state["consecutive_correct"] += 1
            state["consecutive_wrong"] = 0
            if state["consecutive_correct"] >= 3:
                state["difficulty_level"] = min(5, state["difficulty_level"] + 1)
                state["consecutive_correct"] = 0
        else:
            state["consecutive_wrong"] += 1
            state["consecutive_correct"] = 0
            if state["consecutive_wrong"] >= 2:
                state["difficulty_level"] = max(1, state["difficulty_level"] - 1)
                state["consecutive_wrong"] = 0
```

```python
class InterviewAgent:
    MAX_FOLLOW_UPS_PER_ROUND = 2

    async def decide_next(self, state: InterviewState) -> InterviewState:
        """决策：追问 / 下一轮 / 结束面试"""
        prompt = DECIDE_NEXT_PROMPT.format(
            jd=state["jd_content"],
            resume=state["resume_content"],
            current_round=state["current_round"],
            total_rounds=state["total_rounds"],
            current_question=state["current_question"],
            user_answer=state["user_answer"],
            follow_up_count=state["follow_up_count"],
            max_follow_ups=self.MAX_FOLLOW_UPS_PER_ROUND,
            qa_history=self._format_history(state["qa_history"]),
        )
        result = await self.llm.chat(prompt, response_format={"type": "json_object"})
        decision = json.loads(result)

        if state["current_round"] >= state["total_rounds"]:
            # 所有轮次完成 → 生成报告（最后一轮也允许追问一次）
            if (decision.get("should_follow_up")
                    and state["follow_up_count"] < self.MAX_FOLLOW_UPS_PER_ROUND):
                state["next_action"] = "follow_up"
                state["is_follow_up"] = True
                state["follow_up_count"] += 1
            else:
                state["next_action"] = "report"
        elif decision.get("should_follow_up") and \
                state["follow_up_count"] < self.MAX_FOLLOW_UPS_PER_ROUND:
            state["next_action"] = "follow_up"
            state["is_follow_up"] = True
            state["follow_up_count"] += 1
        else:
            state["next_action"] = "next_round"
            state["current_round"] += 1
            state["is_follow_up"] = False
            state["follow_up_count"] = 0

        return state

    async def generate_report(self, state: InterviewState) -> InterviewState:
        """生成完整复盘报告"""
        prompt = GENERATE_REPORT_PROMPT.format(
            jd=state["jd_content"],
            resume=state["resume_content"],
            total_rounds=state["total_rounds"],
            qa_history=self._format_history(state["qa_history"]),
        )
        result = await self.llm.chat(prompt, response_format={"type": "json_object"})
        report = json.loads(result)

        state["report"] = {
            "overall_score": report["overall_score"],
            "dimensions": {
                "tech_depth": report["tech_depth"],
                "clarity": report["clarity"],
                "logic": report["logic"],
                "job_match": report["job_match"],
            },
            "overall_comment": report["overall_comment"],
            "round_reviews": report["round_reviews"],
            "highlights": report["highlights"],
            "weaknesses": report["weaknesses"],
            "suggestions": report["suggestions"],
        }
        state["phase"] = InterviewPhase.DONE
        return state
```

```python
# llm/client.py
import os
from openai import AsyncOpenAI
from typing import AsyncGenerator, Optional, Dict, Any

class UnifiedLLMClient:
    """统一 LLM 客户端，支持 OpenAI 兼容接口的任意模型"""

    def __init__(self):
        self.client = AsyncOpenAI(
            api_key=os.getenv("LLM_API_KEY"),
            base_url=os.getenv("LLM_BASE_URL", "https://open.bigmodel.cn/api/paas/v4/"),
        )
        self.model = os.getenv("LLM_MODEL", "glm-4-flash")
        self.max_retries = 3
        self.timeout = 25.0               # 非流式 25s 超时
        self.stream_timeout = 60.0         # 流式 60s 超时

    async def chat(
        self,
        messages: list[dict],
        temperature: float = 0.7,
        max_tokens: int = 2048,
        response_format: Optional[dict] = None,
    ) -> str:
        """非流式调用"""
        kwargs: Dict[str, Any] = {
            "model": self.model,
            "messages": messages,
            "temperature": temperature,
            "max_tokens": max_tokens,
            "timeout": self.timeout,
        }
        if response_format:
            kwargs["response_format"] = response_format

        for attempt in range(self.max_retries):
            try:
                response = await self.client.chat.completions.create(**kwargs)
                return response.choices[0].message.content
            except Exception as e:
                if attempt == self.max_retries - 1:
                    raise
                await asyncio.sleep(2 ** attempt)

    async def chat_stream(
        self,
        messages: list[dict],
        temperature: float = 0.7,
        max_tokens: int = 2048,
    ) -> AsyncGenerator[str, None]:
        """流式调用，逐 token yield"""
        for attempt in range(self.max_retries):
            try:
                stream = await self.client.chat.completions.create(
                    model=self.model,
                    messages=messages,
                    temperature=temperature,
                    max_tokens=max_tokens,
                    stream=True,
                    timeout=self.stream_timeout,
                )
                async for chunk in stream:
                    delta = chunk.choices[0].delta
                    if delta.content:
                        yield delta.content
                return
            except Exception as e:
                if attempt == self.max_retries - 1:
                    raise
                await asyncio.sleep(2 ** attempt)

    async def chat_with_json(
        self,
        messages: list[dict],
        temperature: float = 0.3,
    ) -> dict:
        """结构化 JSON 输出（简化调用）"""
        text = await self.chat(
            messages=messages,
            temperature=temperature,
            response_format={"type": "json_object"},
        )
        return json.loads(text)
```

## 八、SSE 连接管理器

### 8.1 设计目标

- 每个活跃 session 维护一个 SSE 连接
- Service 层通过 `asyncio.Queue` 向 SSE 连接发送事件
- SSE 连接断线时自动清理 Queue，不阻塞业务逻辑
- 前端重连时通过 `/status` 恢复状态

### 8.2 实现

```python
# sse/manager.py
import asyncio
from typing import Dict, AsyncGenerator
from dataclasses import dataclass, field

@dataclass
class SSEConnection:
    queue: asyncio.Queue = field(default_factory=asyncio.Queue)
    user_id: int = 0

class SSEManager:
    """管理所有活跃的 SSE 连接"""

    def __init__(self):
        self._connections: Dict[str, SSEConnection] = {}

    def create_connection(self, session_key: str, user_id: int) -> SSEConnection:
        """为一个 session 创建 SSE 连接"""
        conn = SSEConnection(user_id=user_id)
        self._connections[session_key] = conn
        return conn

    def remove_connection(self, session_key: str):
        """移除并清理 SSE 连接"""
        conn = self._connections.pop(session_key, None)
        if conn:
            # 放入哨兵值，通知 consumer 关闭
            conn.queue.put_nowait(None)

    async def send_event(
        self, session_key: str, event_type: str, data: str
    ):
        """向指定 session 的 SSE 连接发送事件"""
        conn = self._connections.get(session_key)
        if conn:
            await conn.queue.put({
                "event": event_type,
                "data": data,
            })

    async def stream_events(
        self, session_key: str
    ) -> AsyncGenerator[dict, None]:
        """SSE 事件生成器，供 EventSourceResponse 使用"""
        conn = self._connections.get(session_key)
        if not conn:
            yield {"event": "error", "data": "Session not found"}
            return

        while True:
            try:
                event = await asyncio.wait_for(conn.queue.get(), timeout=30)
                if event is None:  # 哨兵，连接已关闭
                    break
                yield {
                    "event": event["event"],
                    "data": event["data"],
                }
            except asyncio.TimeoutError:
                # 发送心跳
                yield {"event": "ping", "data": "keepalive"}

    def get_active_sessions(self) -> list[str]:
        return list(self._connections.keys())

# 全局单例
sse_manager = SSEManager()
```

### 8.3 SSE 事件发射工具

```python
# sse/emitter.py
import json
from sse.manager import sse_manager

class SSEEmitter:
    """便捷的事件发射工具，封装 JSON 序列化"""

    def __init__(self, session_key: str):
        self.session_key = session_key

    async def emit(self, event_type: str, data: dict):
        await sse_manager.send_event(
            self.session_key,
            event_type,
            json.dumps(data, ensure_ascii=False),
        )

    async def emit_stream_chunks(
        self,
        event_type: str,
        chunks: AsyncGenerator[str, None],
        done_event: str | None = None,
        done_data: dict | None = None,
    ):
        """将 LLM 流式输出逐 chunk 发射为 SSE 事件"""
        full_content = []
        async for chunk in chunks:
            full_content.append(chunk)
            await self.emit(event_type, {"content": chunk})
        if done_event:
            await self.emit(done_event, done_data or {})
        return "".join(full_content)
```

```python
# services/practice_service.py
class PracticeService:
    """
    刷题模式的业务编排层
    负责协调 Agent、SSE、Repository 之间的交互
    """

    def __init__(self, db: AsyncSession, llm: UnifiedLLMClient, sse: SSEManager):
        self.db = db
        self.llm = llm
        self.sse = sse
        self.practice_repo = PracticeRepo(db)
        self.agents: Dict[str, PracticeAgent] = {}  # session_id → agent

    async def create_session(self, user_id: int, topic: str) -> PracticeSession:
        """创建刷题会话 + 校验主题"""
        # 1. 校验主题
        agent = PracticeAgent(self.llm, self.practice_repo)
        state = PracticeState(
            session_id=0, user_id=user_id, topic=topic,
            difficulty_level=3, question_index=0,
            consecutive_correct=0, consecutive_wrong=0,
        )
        state = await agent.validate_topic(state)
        if state.get("error"):
            raise TopicValidationError(state["error"])

        # 2. 创建数据库记录
        session = await self.practice_repo.create_session(
            user_id=user_id, topic=topic, difficulty_level=3
        )
        state["session_id"] = session.id
        self.agents[str(session.id)] = agent
        return session

    async def stream_session(self, session_id: int, user_id: int):
        """SSE 流式主循环"""
        session_key = f"practice:{session_id}"
        self.sse.create_connection(session_key, user_id)

        agent = self.agents.get(str(session_id))
        if not agent:
            # 从 DB 恢复 Agent 状态
            agent = await self._restore_agent(session_id, user_id)

        emitter = SSEEmitter(session_key)

        # 生成题目并流式输出
        state = await agent.graph.aget_state(agent.config)
        state = await agent.generate_question(state)

        # 流式输出题目
        async for chunk in self.llm.chat_stream(
            self._build_question_messages(state)
        ):
            await emitter.emit("question_chunk", {"content": chunk})

        await emitter.emit("question_done", {"question": state["current_question"]})

        # 之后由 submit_answer() 触发后续流转
        # SSE 连接保持，等待 Queue 中的事件

    async def submit_answer(
        self, session_id: int, user_id: int,
        answer: str, time_spent: int
    ):
        """提交答案，触发评估"""
        agent = self.agents[str(session_id)]
        emitter = SSEEmitter(f"practice:{session_id}")

        state = await agent.graph.aget_state(agent.config)
        state["user_answer"] = answer
        state["time_spent"] = time_spent

        # 评估
        state = await agent.evaluate(state)

        # 流式输出反馈
        await emitter.emit_stream_chunks(
            "feedback_chunk",
            self._stream_feedback(state),
            "feedback_done",
            {
                "score": state["score"],
                "is_correct": state["is_correct"],
            },
        )

        # 持久化记录
        await self.practice_repo.create_record(
            session_id=session_id,
            question_number=state["question_index"],
            question=state["current_question"],
            question_type=state["question_type"],
            reference_answer=state["reference_answer"],
            user_answer=answer,
            score=state["score"],
            is_correct=state["is_correct"],
            feedback=state["feedback"],
            knowledge_points=state["knowledge_points"],
            time_spent_seconds=time_spent,
        )

        # 难度调整
        await agent.adjust_difficulty(state)

    async def request_next(self, session_id: int, user_id: int):
        """请求下一题"""
        agent = self.agents[str(session_id)]
        emitter = SSEEmitter(f"practice:{session_id}")

        state = await agent.graph.aget_state(agent.config)

        # 生成新题目（流式输出）
        state = await agent.generate_question(state)
        full_question = await emitter.emit_stream_chunks(
            "question_chunk",
            self.llm.chat_stream(self._build_question_messages(state)),
            "question_done",
            {"question_index": state["question_index"]},
        )
        state["current_question"] = full_question
```

### 10.2 面试服务（InterviewService）

结构与 PracticeService 类似，但状态机更复杂（多了追问分支和报告生成）。

```python
# services/interview_service.py
class InterviewService:
    async def create_session(
        self, user_id: int, jd: str, resume: str, total_rounds: int
    ) -> InterviewSession:
        """创建面试会话"""
        session = await self.interview_repo.create_session(
            user_id=user_id,
            jd_content=jd,
            resume_content=resume,
            total_rounds=total_rounds,
        )
        agent = InterviewAgent(self.llm, self.interview_repo)
        state = InterviewState(
            session_id=session.id, user_id=user_id,
            jd_content=jd, resume_content=resume,
            total_rounds=total_rounds, current_round=1,
            is_follow_up=False, follow_up_count=0,
            qa_history=[], phase=InterviewPhase.ASKING,
        )
        await agent.setup(state)
        self.agents[str(session.id)] = agent
        return session

    async def submit_answer(self, session_id: int, user_id: int, answer: str):
        """提交面试回答 → 决策 → 回应/追问/生成报告"""
        agent = self.agents[str(session_id)]
        emitter = SSEEmitter(f"interview:{session_id}")

        state = await agent.graph.aget_state(agent.config)
        state["user_answer"] = answer

        # 保存 QA 记录
        await self.interview_repo.create_qa(
            session_id=session_id,
            round_number=state["current_round"],
            is_follow_up=state["is_follow_up"],
            question=state["current_question"],
            user_answer=answer,
        )

        # 生成简短回应
        state = await agent.provide_feedback(state)
        await emitter.emit_stream_chunks(
            "feedback_chunk",
            self.llm.chat_stream(self._build_feedback_messages(state)),
            "feedback_done",
            {"action": "pending_decision"},
        )

        # 决策下一步
        state = await agent.decide_next(state)

        if state["next_action"] == "follow_up":
            # 追问：同一轮生成新问题
            state = await agent.ask_question(state)
            await emitter.emit_stream_chunks(
                "question_chunk",
                self.llm.chat_stream(self._build_question_messages(state)),
                "question_done",
                {"round": state["current_round"], "isFollowUp": True},
            )
        elif state["next_action"] == "next_round":
            await emitter.emit("round_advanced", {"round": state["current_round"]})
            # 生成新一轮问题
            state = await agent.ask_question(state)
            await emitter.emit_stream_chunks(
                "question_chunk",
                self.llm.chat_stream(self._build_question_messages(state)),
                "question_done",
                {"round": state["current_round"], "isFollowUp": False},
            )
        elif state["next_action"] == "report":
            # 生成报告
            await emitter.emit("report_generating", {})
            state = await agent.generate_report(state)
            # 持久化报告
            await self.interview_repo.save_report(session_id, state["report"])
            await emitter.emit("report_done", state["report"])
            await emitter.emit("interview_complete", {"sessionId": str(session_id)})
```

```python
# database/connection.py
from sqlalchemy.ext.asyncio import create_async_engine, async_sessionmaker, AsyncSession
from sqlalchemy.orm import DeclarativeBase
from config import settings

engine = create_async_engine(
    settings.database_url,
    echo=settings.debug,
    connect_args={"check_same_thread": False},  # SQLite 异步需要
)

async_session = async_sessionmaker(engine, class_=AsyncSession, expire_on_commit=False)

class Base(DeclarativeBase):
    pass

async def init_db():
    """初始化表结构（开发阶段使用，生产应使用 Alembic 迁移）"""
    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)

async def get_db() -> AsyncGenerator[AsyncSession, None]:
    """FastAPI 依赖注入：获取数据库会话"""
    async with async_session() as session:
        try:
            yield session
            await session.commit()
        except Exception:
            await session.rollback()
            raise
        finally:
            await session.close()
```

```python
# repositories/practice_repo.py
from sqlalchemy import select, func
from sqlalchemy.ext.asyncio import AsyncSession
from models.practice import PracticeSession, PracticeRecord

class PracticeRepo:
    def __init__(self, db: AsyncSession):
        self.db = db

    async def create_session(self, user_id: int, topic: str, difficulty_level: int):
        session = PracticeSession(
            user_id=user_id,
            topic=topic,
            difficulty_level=difficulty_level,
        )
        self.db.add(session)
        await self.db.flush()
        return session

    async def get_session(self, session_id: int) -> PracticeSession | None:
        result = await self.db.execute(
            select(PracticeSession).where(PracticeSession.id == session_id)
        )
        return result.scalar_one_or_none()

    async def create_record(self, **kwargs) -> PracticeRecord:
        record = PracticeRecord(**kwargs)
        self.db.add(record)
        await self.db.flush()
        return record

    async def get_records_by_session(self, session_id: int) -> list[PracticeRecord]:
        result = await self.db.execute(
            select(PracticeRecord)
            .where(PracticeRecord.session_id == session_id)
            .order_by(PracticeRecord.question_number)
        )
        return list(result.scalars().all())

    async def update_session_stats(self, session_id: int, **kwargs):
        session = await self.get_session(session_id)
        if session:
            for key, value in kwargs.items():
                setattr(session, key, value)

    async def get_user_sessions(
        self, user_id: int, page: int = 1, limit: int = 20
    ) -> tuple[list[PracticeSession], int]:
        # 总数
        count_result = await self.db.execute(
            select(func.count()).select_from(PracticeSession)
            .where(PracticeSession.user_id == user_id)
        )
        total = count_result.scalar()

        # 分页
        result = await self.db.execute(
            select(PracticeSession)
            .where(PracticeSession.user_id == user_id)
            .order_by(PracticeSession.created_at.desc())
            .offset((page - 1) * limit)
            .limit(limit)
        )
        return list(result.scalars().all()), total

    async def get_user_stats(self, user_id: int) -> dict:
        """获取用户整体统计数据"""
        sessions = await self.db.execute(
            select(PracticeSession).where(PracticeSession.user_id == user_id)
        )
        sessions_list = list(sessions.scalars().all())

        total_questions = sum(s.total_questions for s in sessions_list)
        total_correct = sum(s.total_correct for s in sessions_list)
        overall_accuracy = total_correct / total_questions if total_questions > 0 else 0

        # 薄弱主题（正确率 < 60% 的 topic）
        weak_topics = []
        for s in sessions_list:
            if s.total_questions > 0:
                accuracy = s.total_correct / s.total_questions
                if accuracy < 0.6:
                    weak_topics.append({
                        "topic": s.topic,
                        "accuracy": round(accuracy * 100),
                        "total_questions": s.total_questions,
                    })

        return {
            "total_questions": total_questions,
            "total_correct": total_correct,
            "overall_accuracy": round(overall_accuracy * 100),
            "weak_topics": weak_topics,
        }
```

## 十七、关键设计决策记录

| 决策 | 选择 | 理由 |
|------|------|------|
| Agent 状态存储 | 内存 + LangGraph checkpointer | 单机单用户场景无需 Redis；checkpointer 用 SQLite 即可支持断点续传 |
| SSE vs WebSocket | SSE | 单向流（服务端→客户端）即可满足需求；SSE 实现更简单，HTTP 原生兼容 |
| Worker 数量 | 1 | 多 worker 时内存中的 Agent 实例和 SSE 连接无法共享；单用户场景无并发压力 |
| ORM vs 原生 SQL | SQLAlchemy 2.0 | ORM 对小型项目更友好；异步会话支持良好 |
| 结构化输出 | JSON mode（而非 function calling） | Prompt 更可控；国内模型 function calling 实现差异较大 |
| 题目生成方式 | 纯 LLM（不用 RAG） | MVP 阶段优先验证核心流程；后续可引入题库 + RAG 提升题目质量 |
