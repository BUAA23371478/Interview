- 缁村害鍦ㄨ繍琛屾椂浠?API 鍝嶅簲鎺㈡祴

from __future__ import annotations

import hashlib
import struct
from typing import List, Optional

import httpx
from loguru import logger

from app.config import settings

MOCK_DIM = 256


def _mock_embed(text: str) -> List[float]:
    """纭畾鎬?mock锛氫粠鏂囨湰鍝堝笇娲剧敓涓€涓?256 缁村崟浣嶅悜閲忋€?""
    digest = hashlib.sha256(text.encode("utf-8")).digest()
    vec = [float(b) for b in digest]  # 32 瀛楄妭
    # 鎵╁埌 256 缁达細鐢?digest 浣滀负绉嶅瓙鍋氬娆℃墿灞?    while len(vec) < MOCK_DIM:
        digest = hashlib.sha256(digest + text.encode("utf-8")).digest()
        vec.extend(float(b) for b in digest)
    vec = vec[:MOCK_DIM]
    norm = sum(v * v for v in vec) ** 0.5 or 1.0
    return [v / norm for v in vec]


class EmbeddingClient:
    """缁熶竴宓屽叆瀹㈡埛绔€?""

    def __init__(self) -> None:
        self._api_key = settings.embedding_api_key or settings.llm_api_key
        self._base_url = settings.embedding_base_url
        self._model = settings.embedding_model
        self._dimension: Optional[int] = None
        self._api_failed = False

    @property
    def enabled(self) -> bool:
        return bool(self._api_key) and not self._api_failed

    async def embed(self, text: str) -> List[float]:
        if not self._api_key or self._api_failed:
            return _mock_embed(text)
        try:
            async with httpx.AsyncClient(timeout=30.0) as client:
                resp = await client.post(
                    f"{self._base_url}/embeddings",
                    headers={"Authorization": f"Bearer {self._api_key}"},
                    json={"model": self._model, "input": text},
                )
                resp.raise_for_status()
                data = resp.json()["data"][0]["embedding"]
                self._dimension = len(data)
                return data
        except Exception as e:  # noqa: BLE001
            self._api_failed = True
            logger.warning("Embedding API 澶辫触锛屽垏鎹㈠埌 mock锛歿}", e)
            return _mock_embed(text)

    async def embed_batch(self, texts: List[str]) -> List[List[float]]:
        if not self._api_key or self._api_failed:
            return [_mock_embed(t) for t in texts]
        try:
            async with httpx.AsyncClient(timeout=60.0) as client:
                resp = await client.post(
                    f"{self._base_url}/embeddings",
                    headers={"Authorization": f"Bearer {self._api_key}"},
                    json={"model": self._model, "input": texts},
                )
                resp.raise_for_status()
                items = sorted(resp.json()["data"], key=lambda x: x["index"])
                vectors = [it["embedding"] for it in items]
                if vectors:
                    self._dimension = len(vectors[0])
                return vectors
        except Exception as e:  # noqa: BLE001
            self._api_failed = True
            logger.warning("Embedding batch API 澶辫触锛屽垏鎹㈠埌 mock锛歿}", e)
            return [_mock_embed(t) for t in texts]

    def test_connection(self) -> dict:
        if not self._api_key:
            return {"ok": False, "model": self._model, "dimension": MOCK_DIM, "error": "鏈厤缃?API Key锛岃繍琛屽湪 mock 妯″紡"}
        if self._api_failed:
            return {"ok": False, "model": self._model, "dimension": MOCK_DIM, "error": "API 璋冪敤澶辫触锛屽凡闄嶇骇 mock"}
        return {"ok": True, "model": self._model, "dimension": self._dimension}


def pack_vector(vec: List[float]) -> bytes:
    """鍚戦噺 鈫?BLOB銆?""
    return struct.pack(f">{len(vec)}d", *vec)


def unpack_vector(blob: bytes) -> List[float]:
    """BLOB 鈫?鍚戦噺銆?""
    n = len(blob) // 8
    return list(struct.unpack(f">{n}d", blob))


def cosine_similarity(a: List[float], b: List[float]) -> float:
    n = min(len(a), len(b))
    dot = sum(a[i] * b[i] for i in range(n))
    na = sum(v * v for v in a[:n]) ** 0.5 or 1.0
    nb = sum(v * v for v in b[:n]) ** 0.5 or 1.0
    return dot / (na * nb)
